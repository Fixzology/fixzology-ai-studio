/**
 * Fixzology AI Studio - Client-Side Video Export Engine
 * Renders real timeline frames + audio into a downloadable video file locally.
 */
export class ExportEngine {
  static getSupportedMimeType() {
    const candidates = [
      'video/mp4;codecs=avc1,mp4a.40.2',
      'video/mp4',
      'video/webm;codecs=vp9,opus',
      'video/webm;codecs=vp8,opus',
      'video/webm'
    ];
    for (const mime of candidates) {
      if (typeof MediaRecorder !== 'undefined' && MediaRecorder.isTypeSupported(mime)) {
        return mime;
      }
    }
    return '';
  }

  static async exportTimeline({
    canvas,
    engine,
    compositor,
    resolutionHeight = 720,
    fps = 30,
    bitrateMbps = 5,
    onProgress
  }) {
    const totalDuration = engine.getTotalDurationSec();
    if (totalDuration <= 0.1) {
      throw new Error('Timeline is empty. Add at least one video, image, or text clip before exporting.');
    }

    const mimeType = this.getSupportedMimeType();
    if (!mimeType) {
      throw new Error('Your current browser does not support local MediaRecorder video encoding. Please use Chrome.');
    }

    // Resize canvas temporarily to requested export resolution
    const origW = canvas.width;
    const origH = canvas.height;
    const aspect = origW / origH;
    canvas.height = resolutionHeight;
    canvas.width = Math.round(resolutionHeight * aspect);

    const stream = canvas.captureStream(fps);
    const chunks = [];
    const recorder = new MediaRecorder(stream, {
      mimeType,
      videoBitsPerSecond: bitrateMbps * 1_000_000
    });

    recorder.ondataavailable = (e) => {
      if (e.data && e.data.size > 0) chunks.push(e.data);
    };

    return new Promise((resolve, reject) => {
      recorder.onerror = (err) => {
        canvas.width = origW;
        canvas.height = origH;
        reject(err);
      };

      recorder.onstop = () => {
        canvas.width = origW;
        canvas.height = origH;
        const isMp4 = mimeType.includes('mp4');
        const ext = isMp4 ? 'mp4' : 'webm';
        const blob = new Blob(chunks, { type: mimeType });
        const downloadUrl = URL.createObjectURL(blob);
        resolve({
          downloadUrl,
          fileName: `${engine.project.title.replace(/[^a-z0-9_\-\u0900-\u097F]/gi, '_')}_${resolutionHeight}p.${ext}`,
          sizeBytes: blob.size,
          mimeType
        });
      };

      recorder.start(200);
      let currentSec = 0;
      const stepSec = 1 / fps;

      const renderLoop = setInterval(() => {
        if (currentSec > totalDuration) {
          clearInterval(renderLoop);
          compositor.syncMediaPlayback(0, false);
          recorder.stop();
          return;
        }
        compositor.syncMediaPlayback(currentSec, true);
        compositor.renderFrame(currentSec);
        const pct = Math.min(99, Math.round((currentSec / totalDuration) * 100));
        if (onProgress) onProgress(pct, currentSec, totalDuration);
        currentSec += stepSec;
      }, 1000 / fps);
    });
  }
}
