/**
 * Fixzology AI Studio - Phase 2 Non-Destructive Timeline & Clip Properties Engine
 */
export class TimelineEngine {
  constructor() {
    this.project = this.createEmptyProject('Fixzology Phase 2 Edit');
    this.undoStack = [];
    this.redoStack = [];
    this.mediaElements = new Map(); // assetId -> { url, element, durationSec, waveform }
  }

  createEmptyProject(title) {
    return {
      schemaVersion: '2.0.0',
      id: 'proj_' + Date.now(),
      title,
      aspectRatio: '16:9',
      width: 1280,
      height: 720,
      fps: 30,
      updatedAt: new Date().toISOString(),
      assets: [],
      tracks: [
        { id: 'track_text', name: 'Text / हिंदी', type: 'text', locked: false, muted: false },
        { id: 'track_video_1', name: 'Video Track 1', type: 'video', locked: false, muted: false },
        { id: 'track_audio_1', name: 'Audio / Voice', type: 'audio', locked: false, muted: false }
      ],
      clips: []
    };
  }

  snapshot() {
    this.undoStack.push(JSON.stringify(this.project));
    if (this.undoStack.length > 40) this.undoStack.shift();
    this.redoStack = [];
  }

  undo() {
    if (!this.undoStack.length) return false;
    this.redoStack.push(JSON.stringify(this.project));
    this.project = JSON.parse(this.undoStack.pop());
    return true;
  }

  redo() {
    if (!this.redoStack.length) return false;
    this.undoStack.push(JSON.stringify(this.project));
    this.project = JSON.parse(this.redoStack.pop());
    return true;
  }

  defaultClipProps() {
    return {
      scale: 1,
      rotation: 0,
      posX: 0,
      posY: 0,
      opacity: 1,
      cropZoom: 1,
      speed: 1,
      volume: 1,
      brightness: 100,
      contrast: 100,
      saturation: 100,
      transitionIn: 'none'
    };
  }

  async importLocalFile(file) {
    this.snapshot();
    const assetId = 'asset_' + Date.now() + '_' + Math.floor(Math.random() * 1000);
    const blobUrl = URL.createObjectURL(file);

    const isAudio = file.type.startsWith('audio/');
    const isImage = file.type.startsWith('image/');
    const mediaType = isAudio ? 'audio' : isImage ? 'image' : 'video';
    const targetTrackId = isAudio ? 'track_audio_1' : 'track_video_1';

    let detectedDuration = 5.0;
    let element = null;

    if (mediaType === 'video') {
      element = document.createElement('video');
      element.src = blobUrl;
      element.crossOrigin = 'anonymous';
      element.preload = 'auto';
      element.playsInline = true;
      await new Promise((res) => {
        element.onloadedmetadata = () => {
          detectedDuration = Number.isFinite(element.duration) && element.duration > 0 ? element.duration : 5.0;
          res();
        };
        element.onerror = () => res();
      });
    } else if (mediaType === 'audio') {
      element = document.createElement('audio');
      element.src = blobUrl;
      element.preload = 'auto';
      await new Promise((res) => {
        element.onloadedmetadata = () => {
          detectedDuration = Number.isFinite(element.duration) && element.duration > 0 ? element.duration : 5.0;
          res();
        };
        element.onerror = () => res();
      });
    } else if (mediaType === 'image') {
      element = new Image();
      element.src = blobUrl;
      await new Promise((res) => {
        element.onload = () => res();
        element.onerror = () => res();
      });
    }

    this.mediaElements.set(assetId, { url: blobUrl, element, durationSec: detectedDuration });

    this.project.assets.push({
      id: assetId,
      name: file.name,
      type: mediaType,
      sizeBytes: file.size,
      durationSec: Number(detectedDuration.toFixed(2))
    });

    const trackEnd = this.project.clips
      .filter((c) => c.trackId === targetTrackId)
      .reduce((max, c) => Math.max(max, c.startSec + c.durationSec), 0);

    const clip = {
      id: 'clip_' + Date.now() + '_' + Math.floor(Math.random() * 1000),
      assetId,
      trackId: targetTrackId,
      label: file.name,
      type: mediaType,
      startSec: Number(trackEnd.toFixed(2)),
      durationSec: Number(detectedDuration.toFixed(2)),
      inPointSec: 0,
      color: isAudio ? '#00D2D3' : '#7C5CFF',
      props: this.defaultClipProps()
    };

    this.project.clips.push(clip);
    return clip;
  }

  addTextClip(textContent, startSec) {
    this.snapshot();
    const clip = {
      id: 'txt_' + Date.now(),
      trackId: 'track_text',
      label: textContent.slice(0, 22),
      type: 'text',
      textContent,
      textColor: '#FFFFFF',
      fontSize: 42,
      startSec: Number(startSec.toFixed(2)),
      durationSec: 4.0,
      color: '#3B82F6',
      props: this.defaultClipProps()
    };
    this.project.clips.push(clip);
    return clip;
  }

  updateClipProps(clipId, partialProps) {
    const clip = this.project.clips.find((c) => c.id === clipId);
    if (!clip) return;
    clip.props = { ...clip.props, ...partialProps };
    if (partialProps.speed && partialProps.speed > 0) {
      const asset = this.project.assets.find((a) => a.id === clip.assetId);
      if (asset) {
        clip.durationSec = Number((asset.durationSec / partialProps.speed).toFixed(2));
      }
    }
  }

  splitClipAt(clipId, playheadSec) {
    const idx = this.project.clips.findIndex((c) => c.id === clipId);
    if (idx === -1) return false;
    const clip = this.project.clips[idx];
    const track = this.project.tracks.find((t) => t.id === clip.trackId);
    if (track?.locked) return false;

    const offset = playheadSec - clip.startSec;
    if (offset <= 0.15 || offset >= clip.durationSec - 0.15) return false;

    this.snapshot();
    const firstHalf = { ...clip, durationSec: Number(offset.toFixed(2)), props: { ...clip.props } };
    const secondHalf = {
      ...clip,
      id: 'clip_' + Date.now(),
      startSec: Number(playheadSec.toFixed(2)),
      durationSec: Number((clip.durationSec - offset).toFixed(2)),
      inPointSec: Number(((clip.inPointSec || 0) + offset * (clip.props.speed || 1)).toFixed(2)),
      props: { ...clip.props }
    };

    this.project.clips.splice(idx, 1, firstHalf, secondHalf);
    return true;
  }

  deleteClip(clipId) {
    const clip = this.project.clips.find((c) => c.id === clipId);
    if (!clip) return false;
    const track = this.project.tracks.find((t) => t.id === clip.trackId);
    if (track?.locked) return false;

    this.snapshot();
    this.project.clips = this.project.clips.filter((c) => c.id !== clipId);
    return true;
  }

  getTotalDurationSec() {
    return this.project.clips.reduce((max, c) => Math.max(max, c.startSec + c.durationSec), 0);
  }
}
